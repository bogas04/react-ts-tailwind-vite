const TestFnOne = () =>
  new Promise((resolve) => {
    const output = [];

    for (let i = 0; i < 30000; i++) {
      output.push({ id: i, randomNumber: Math.round(Math.random() * 1000) });
    }

    resolve(output);
  });

const TestFnTwo = () => {
  const output = new Array(50000).fill(null);

  for (let i = 0; i < output.length; i++) {
    output[i] = { id: i, randomNumber: Math.round(Math.random() * 1000) };
  }

  return output;
};

const TestFnThree = () =>
  new Promise((resolve) => {
    const output = [];

    for (let i = 0; i < 100000; i++) {
      output.push({ id: i, randomNumber: Math.round(Math.random() * 1000) });
    }

    resolve(output);
  });

const TestFnSlow = () =>
  new Promise((resolve) => {
    const isSlow = Boolean(Math.random() <= 0.25);

    const delay = isSlow ? 50 : 5 + Math.random() * 5;

    setTimeout(resolve, delay);
  });

async function checkPerformance(fn: VoidFunction) {
  let timeMS = 0;

  const before = performance.now();

  await fn();

  timeMS = performance.now() - before;

  return timeMS;
}

function testOnce() {
  const fns = [() => TestFnOne(), () => TestFnTwo(), () => TestFnThree()];

  Promise.all(fns.map((fn) => checkPerformance(fn))).then((results) =>
    console.log(...results)
  );
}

async function benchmark(runs = 30, fn: any, mode = "avg", threshold = 15) {
  const results = [];

  for (let i = 0; i < runs; i++) {
    const timeMs = await Promise.race([
      checkPerformance(fn),
      delay(threshold + 1),
    ]);

    if (mode === "avg") {
      if (timeMs < threshold) {
        results.push(timeMs);
      }
    }
  }

  return results.reduce((acc, item) => acc + item, 0) / results.length;
}

async function benchmarkMany(runs = 30, mode = "avg", fns: any[]) {
  const avgs = [];

  for (let fnIndex = 0; fnIndex < fns.length; fnIndex++) {
    const fn = fns[fnIndex];
    avgs.push({
      index: fnIndex,
      name: fn.name,
      avg: await benchmark(runs, fn, mode),
    });
  }

  return avgs.toSorted((a, b) => a.avg - b.avg);
}

(async () => {
  console.log(
    await benchmarkMany(30, "avg", [
      TestFnTwo,
      TestFnOne,
      TestFnThree,
      TestFnSlow,
    ])
  );
})();

function delay(ms: number): Promise<number> {
  return new Promise((resolve) => setTimeout(resolve, ms, ms));
}
